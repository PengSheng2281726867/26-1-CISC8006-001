from pathlib import Path
import sys
sys.path.insert(0,str(Path(__file__).resolve().parents[1]))
from src.paths import ROOT,save_json,tensor_hash
from src.factory import build_model as frozen_model,seed_all,configure_determinism
from src.train import train_model
from ablation_mlp.factory import build_model
from ablation_mlp.model import canonical_scorer_state
import copy
import json
import unittest
import torch
from torch import nn
from torch_geometric.data import Data

torch.set_num_threads(1);configure_determinism();EVIDENCE=[]

def graph():
    seed_all(30)
    return Data(x=torch.randn(8,5),edge_index=torch.tensor([[0,1,2,3,4,5,6,7,1,4],[1,2,3,4,5,6,7,0,4,1]]),
        y=torch.tensor([0,1,2,0,1,2,0,1]),train_mask=torch.tensor([1,1,1,1,0,0,0,0],dtype=torch.bool),
        val_mask=torch.tensor([0,0,0,0,1,1,0,0],dtype=torch.bool),test_mask=torch.tensor([0,0,0,0,0,0,1,1],dtype=torch.bool))

class ComponentTests(unittest.TestCase):
    def test_frozen_baseline_forward_and_gradients(self):
        for device in ('cpu','cuda'):
            for k in (1,2,3,4):
                for training in (False,True):
                    old,om=frozen_model(5,3,k,'mlp',42,device)
                    new,nm=build_model(5,3,k,'mlp',42,device)
                    self.assertEqual(om['full_init_sha256'],nm['full_init_sha256'])
                    old.train(training);new.train(training)
                    x=graph().x.to(device);edge=graph().edge_index.to(device)
                    seed_all(99);a=old(x,edge);a.square().sum().backward()
                    seed_all(99);b=new(x,edge);b.square().sum().backward()
                    torch.testing.assert_close(a,b,rtol=0,atol=0)
                    for key,p in old.named_parameters():
                        q=dict(new.named_parameters())[key]
                        if p.grad is None:self.assertIsNone(q.grad)
                        else:torch.testing.assert_close(p.grad,q.grad,rtol=0,atol=0)
                    EVIDENCE.append(dict(test='baseline_alignment',device=device,k=k,training=training,max_error=0))

    def test_single_component_and_matching_initialization(self):
        expected_linear=None
        for k in (1,2,3,4):
            base,bm=build_model(5,3,k,'mlp',42,'cpu')
            for variant in ('mlp_ln','mlp_gelu'):
                model,meta=build_model(5,3,k,variant,42,'cpu');scorer=model.convs[0].scorer
                self.assertEqual(meta['common_init_sha256'],bm['common_init_sha256'])
                self.assertEqual(meta['full_common_init_sha256'],bm['full_common_init_sha256'])
                self.assertEqual(meta['cpu_rng_sha256'],bm['cpu_rng_sha256'])
                self.assertEqual(meta['common_scorer_linear_init_sha256'],bm['common_scorer_linear_init_sha256'])
                expected_linear=expected_linear or meta['common_scorer_linear_init_sha256']
                self.assertEqual(expected_linear,meta['common_scorer_linear_init_sha256'])
                types=[type(module) for module in scorer]
                if variant=='mlp_ln':
                    self.assertEqual(types,[nn.Linear,nn.LayerNorm,nn.ReLU,nn.Linear])
                    self.assertEqual(scorer[1].normalized_shape,(32,));self.assertEqual(scorer[1].eps,1e-5)
                    torch.testing.assert_close(scorer[1].weight,torch.ones(32));torch.testing.assert_close(scorer[1].bias,torch.zeros(32))
                else:self.assertEqual(types,[nn.Linear,nn.GELU,nn.Linear])
                self.assertEqual(sum(p.numel() for p in model.parameters())-sum(p.numel() for p in base.parameters()),64 if variant=='mlp_ln' else 0)
                self.assertEqual(model.convs[0].powers,list(range(k+1)));self.assertEqual(len(model.convs),1)

    def test_shapes_gradients_routing_and_fresh_logits(self):
        for device in ('cpu','cuda'):
            for variant in ('mlp_ln','mlp_gelu'):
                for k in (1,2,3,4):
                    g=graph().to(device);model,meta=build_model(5,3,k,variant,42,device);model.eval()
                    captured=[]
                    hook=model.convs[0].scorer[0].register_forward_pre_hook(lambda mod,inputs:captured.append(list(inputs[0].shape)))
                    out,details=model(g.x,g.edge_index,return_diagnostics=True);hook.remove()
                    self.assertEqual(captured,[[8,k+1,64]]*2);self.assertEqual(list(out.shape),[8,3])
                    layer=details[0];score=layer['iterations'][0]['agreement'];score.retain_grad()
                    self.assertEqual(list(score.shape),[8,k+1])
                    torch.testing.assert_close(layer['used_alpha'],torch.softmax(score,dim=1))
                    for step in layer['iterations']:
                        for value in step.values():self.assertTrue(bool(torch.isfinite(value).all()))
                    torch.nn.functional.nll_loss(out[g.train_mask],g.y[g.train_mask]).backward()
                    self.assertGreater(float(score.grad.abs().max()),0)
                    grads={}
                    for name,p in model.named_parameters():
                        if p.grad is not None:self.assertTrue(bool(torch.isfinite(p.grad).all()))
                        if '.scorer.' in name and name.endswith('weight'):
                            self.assertGreater(float(p.grad.norm()),0);grads[name]=float(p.grad.norm())
                    second,d2=model(g.x,g.edge_index,return_diagnostics=True)
                    torch.testing.assert_close(out,second,rtol=0,atol=0)
                    self.assertEqual(float(d2[0]['iterations'][0]['logits'].abs().max()),0)
                    model.zero_grad();model.convs[0].num_iterations=1
                    model(g.x,g.edge_index).sum().backward()
                    self.assertTrue(all(p.grad is None for p in model.convs[0].scorer.parameters()))
                    EVIDENCE.append(dict(test='gradient',device=device,method=variant,k=k,grads=grads))

    def test_test_labels_isolated(self):
        config=json.loads((ROOT/'configs/main.json').read_text());config['epochs']=8
        for variant in ('mlp_ln','mlp_gelu'):
            g=graph();poison=g.clone();poison.y[poison.test_mask]=(poison.y[poison.test_mask]+1)%3
            a,_=build_model(5,3,2,variant,42,'cpu');b=copy.deepcopy(a)
            seed_all(97);ra=train_model(a,g,config)
            seed_all(97);rb=train_model(b,poison,config)
            self.assertEqual(tensor_hash(a.state_dict()),tensor_hash(b.state_dict()))
            self.assertEqual(ra['best_validation_epoch'],rb['best_validation_epoch'])
            self.assertEqual([r['train_loss'] for r in ra['history']],[r['train_loss'] for r in rb['history']])

if __name__=='__main__':
    result=unittest.TextTestRunner(verbosity=2).run(unittest.defaultTestLoader.loadTestsFromTestCase(ComponentTests))
    save_json(ROOT/'results/mlp_components_unit_tests.json',dict(passed=result.wasSuccessful(),tests_run=result.testsRun,
        errors=len(result.errors),failures=len(result.failures),evidence=EVIDENCE))
    sys.exit(0 if result.wasSuccessful() else 1)
