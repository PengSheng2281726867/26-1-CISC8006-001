from pathlib import Path
import sys
sys.path.insert(0,str(Path(__file__).resolve().parents[1]))

from ablation_mlp.factory import build_model
import argparse

import traceback

GROUP='mlp_components_v1'
METHODS=('mlp_ln','mlp_gelu')

def read(path):return json.loads(Path(path).read_text(encoding='utf-8'))

def signature():
    previous=read(ROOT/'runs/v2/protocol.json')
    for relative,digest in previous['sources'].items():assert sha256(ROOT/relative)==digest
    assert sha256(ROOT/'data/manifest.json')==previous['data_manifest_sha256']
    sources={relative:sha256(ROOT/relative) for relative in
        ('ablation_mlp/model.py','ablation_mlp/factory.py','ablation_mlp/run.py')}
    content=dict(parent_protocol_sha256=previous['sha256'],parent_sources=previous['sources'],
        sources=sources,training=read(ROOT/'configs/main.json'),data_manifest_sha256=sha256(ROOT/'data/manifest.json'),
        methods={'mlp_ln':'Linear64-32, LayerNorm32 eps1e-5 affine, ReLU, Linear32-1',
                 'mlp_gelu':'Linear64-32, GELU exact, Linear32-1'},
        common_initialization='Both MLP linears and all shared GAMMA weights match v2 MLP; norm gamma1 beta0')
    fingerprint=hashlib.sha256(json.dumps(content,sort_keys=True).encode()).hexdigest()
    return fingerprint,content

def complete(folder,fingerprint=None):
    path=folder/'result.json'
    if not path.exists():return None
    report=read(path)
    if report.get('status')!='complete':return None
    if fingerprint is not None:assert report['protocol_sha256']==fingerprint,'Frozen protocol mismatch'
    artifacts=folder/report['artifact_subdir']
    if not all((artifacts/name).is_file() and sha256(artifacts/name)==digest for name,digest in report['artifacts'].items()):return None
    return report

def baseline(name,k,split):
    folder=ROOT/'runs/v2/full'/name.lower()/f'k_{k}'/f'split_{split}'/'mlp'
    report=complete(folder,read(ROOT/'runs/v2/protocol.json')['sha256'])
    assert report is not None
    return folder,report

def train_one(name,dataset,graph,k,split,method,phase,folder,fingerprint,config):
    reference_folder,reference=baseline(name,k,split)
    masks=load_masks(name,split,graph.num_nodes)
    for label,mask in zip(('train_mask','val_mask','test_mask'),masks):graph[label]=torch.from_numpy(mask).to('cuda')
    folder.mkdir(parents=True,exist_ok=True)
    report=complete(folder,fingerprint)
    if report is not None:return report
    for attempt in range(1,4):
        output=folder/f'attempt_{attempt:02d}'
        if (output/'failure.json').exists():continue
        output.mkdir(exist_ok=True);model=None;handles=[]
        try:
            torch.cuda.synchronize();gc.collect();torch._C._cuda_clearCublasWorkspaces();torch.cuda.empty_cache()
            model,initial=build_model(dataset.num_node_features,dataset.num_classes,k,method,42+split)
            assert initial['reference_full_init_sha256']==reference['full_init_sha256']
            for key in ('common_init_sha256','cpu_rng_sha256','cuda_rng_sha256'):assert initial[key]==reference[key]
            assert len(model.convs)==1 and model.convs[0].powers==list(range(k+1)) and model.convs[0].num_iterations==2
            gradients={}
            def capture(key):
                def hook(value):
                    if key not in gradients:
                        norm=float(value.norm());assert np.isfinite(norm) and norm>0,key
                        gradients[key]=norm
                    return value
                return hook
            for key,param in model.named_parameters():
                if '.scorer.' in key and key.endswith('weight'):handles.append(param.register_hook(capture(key)))
            meta=dict(dataset=name,routing_method=method,k=k,powers=list(range(k+1)),actual_gamma_layers=1,r=2,
                seed=42+split,split_id=split,phase=phase,group=GROUP,protocol_sha256=fingerprint,
                split_sha256=sha256(split_path(name,split)),reference_run=str(reference_folder.relative_to(ROOT)),
                reference_trained_state_sha256=reference['trained_state_sha256'],scorer_weight_gradient_first=gradients,**initial)
            assert meta['split_sha256']==reference['split_sha256']
            report=run_one(model,graph,config,meta,output)
            assert report['scorer_parameter_count']==(2177 if method=='mlp_ln' else 2113)
            assert report['scorer_gradient_norm_first']>0 and report['scorer_parameter_change_l2']>0
            assert len(gradients)==(3 if method=='mlp_ln' else 2)
            if phase=='anchors':
                assert method=='mlp'
                assert report['trained_state_sha256']==reference['trained_state_sha256']
                assert report['best_validation_epoch']==reference['best_validation_epoch'] and report['stopped_epoch']==reference['stopped_epoch']
                with np.load(output/'predictions.npz') as current,np.load(reference_folder/reference['artifact_subdir']/'predictions.npz') as old:
                    for key in ('predictions','log_probs','alpha'):np.testing.assert_array_equal(current[key],old[key])
                report['baseline_replay_exact']=True
            report['artifact_subdir']=output.name
            save_json(folder/'result.json',report)
            return report
        except Exception as error:
            save_json(output/'failure.json',dict(error=repr(error),traceback=traceback.format_exc(),
                dataset=name,k=k,split=split,method=method,phase=phase,attempt=attempt))
            print(f'FAILED {name} K{k} split{split} {method} attempt{attempt}: {error!r}',flush=True)
        finally:
            for handle in handles:handle.remove()
            del model;gc.collect();torch.cuda.empty_cache()
    raise RuntimeError(f'Exhausted retries: {folder}')

def main():
    parser=argparse.ArgumentParser();parser.add_argument('--phase',choices=['anchors','pilot','full'],required=True)
    args=parser.parse_args();phase=args.phase
    assert read(ROOT/'results/mlp_components_unit_tests.json')['passed']
    fingerprint,content=signature();group=ROOT/'runs'/GROUP;group.mkdir(exist_ok=True)
    protocol=group/'protocol.json'
    if protocol.exists():assert read(protocol)['sha256']==fingerprint
    else:
        save_json(protocol,dict(sha256=fingerprint,**content))
        snapshot=group/'source_snapshot'
        for relative in list(content['sources'])+list(content['parent_sources']):
            target=snapshot/relative;target.parent.mkdir(parents=True,exist_ok=True);shutil.copy2(ROOT/relative,target)
        (group/'pip_freeze.txt').write_bytes(subprocess.check_output([sys.executable,'-B','-m','pip','freeze']))
        save_json(group/'environment.json',dict(python=sys.version,torch=torch.__version__,pyg=torch_geometric.__version__,
            cuda=torch.version.cuda,gpu=torch.cuda.get_device_name(),numpy=np.__version__,executable=sys.executable))
    config=read(ROOT/'configs/main.json')
    specs=dict(anchors=dict(datasets=['Texas','Chameleon'],ks=[2],splits=[0],methods=['mlp']),
        pilot=dict(datasets=['Texas','Chameleon'],ks=[2,3,4],splits=[0,1],methods=list(METHODS)),
        full=dict(datasets=config['datasets'],ks=[1,2,3,4],splits=list(range(10)),methods=list(METHODS)))
    specification=specs[phase]
    if phase!='anchors':assert read(group/'anchors/matrix_status.json')['status']=='complete'
    if phase=='full':assert read(group/'pilot/verification.json')['status']=='passed'
    configure_determinism();start=time.perf_counter();phase_dir=group/phase;phase_dir.mkdir(exist_ok=True)
    save_json(group/f'config_{phase}.json',dict(**config,component_specification=specification))
    expected=int(np.prod([len(specification[key]) for key in ('datasets','ks','splits','methods')]))
    status=dict(status='running',phase=phase,expected=expected,completed=0,reused_pilot=0,
        protocol_sha256=fingerprint,started_at=datetime.now(timezone.utc).isoformat(),failed=[])
    log=(phase_dir/'scheduler.log').open('a',encoding='utf-8')
    try:
        for name in specification['datasets']:
            dataset=load_dataset(name);graph=dataset[0].to('cuda')
            for k in specification['ks']:
                for split in specification['splits']:
                    order=specification['methods'] if split%2==0 else list(reversed(specification['methods']))
                    for method in order:
                        folder=phase_dir/name.lower()/f'k_{k}'/f'split_{split}'/method
                        report=complete(folder,fingerprint)
                        if report is None and phase=='full':
                            pilot=group/'pilot'/name.lower()/f'k_{k}'/f'split_{split}'/method
                            reusable=complete(pilot,fingerprint)
                            if reusable is not None:
                                shutil.copytree(pilot,folder,dirs_exist_ok=True)
                                report={**reusable,'phase':'full','reused_from':str(pilot.relative_to(ROOT))}
                                save_json(folder/'result.json',report);status['reused_pilot']+=1
                        if report is None:
                            report=train_one(name,dataset,graph,k,split,method,phase,folder,fingerprint,config)
                        status['completed']+=1;save_json(phase_dir/'matrix_status.json',status)
                        message=f"DONE {status['completed']}/{expected} {name} K{k} split{split} {method} test={100*report['test_accuracy']:.2f}% epochs={report['stopped_epoch']}"
                        print(message,flush=True);log.write(message+'\n');log.flush()
            del graph,dataset;gc.collect();torch.cuda.empty_cache()
        status.update(status='complete',elapsed_seconds=time.perf_counter()-start,finished_at=datetime.now(timezone.utc).isoformat())
        save_json(phase_dir/'matrix_status.json',status);print(f'FINISHED {phase} {expected}/{expected}',flush=True)
    except Exception as error:
        status.update(status='partial',error=repr(error),elapsed_seconds=time.perf_counter()-start)
        save_json(phase_dir/'matrix_status.json',status);raise
    finally:log.close()

if __name__=='__main__':main()
