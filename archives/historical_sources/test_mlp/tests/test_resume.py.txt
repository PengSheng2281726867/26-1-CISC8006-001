import sys
from pathlib import Path
sys.path.insert(0,str(Path(__file__).resolve().parents[1]))
from src.paths import ROOT,sha256,save_json
from scripts.run_experiments import completed_run
import unittest

class ResumeTests(unittest.TestCase):
    def test_complete_missing_corrupt_and_protocol_mismatch(self):
        folder=ROOT/'cache/resume_unit';artifacts=folder/'attempt_01';artifacts.mkdir(parents=True,exist_ok=True)
        for name in ('model.pt','predictions.npz','epochs.csv'): (artifacts/name).write_text('test artifact',encoding='utf-8')
        report=dict(status='complete',protocol_sha256='expected',artifact_subdir='attempt_01',
                    artifacts={name:sha256(artifacts/name) for name in ('model.pt','predictions.npz','epochs.csv')})
        save_json(folder/'result.json',report)
        self.assertIsNotNone(completed_run(folder,'expected'))
        with self.assertRaises(RuntimeError):completed_run(folder,'other protocol')
        (artifacts/'epochs.csv').write_text('corruption',encoding='utf-8')
        self.assertIsNone(completed_run(folder,'expected'))
        self.assertIsNone(completed_run(ROOT/'cache/missing_resume_test','expected'))

if __name__=='__main__':
    result=unittest.TextTestRunner(verbosity=2).run(unittest.defaultTestLoader.loadTestsFromTestCase(ResumeTests))
    save_json(ROOT/'results/resume_unit_tests.json',dict(passed=result.wasSuccessful(),tests_run=result.testsRun))
    sys.exit(0 if result.wasSuccessful() else 1)
