def read(path):return json.loads(Path(path).read_text(encoding='utf-8'))
def signed(value):return '0.00' if abs(value)<1e-9 else f'{value:+.2f}'
def table(headers,rows):
    return '| '+' | '.join(headers)+' |\n| '+' | '.join(['---']*len(headers))+' |\n'+''.join('| '+' | '.join(map(str,r))+' |\n' for r in rows)

def main():

    pairs={(r['dataset'],r['k'],r['method']):r for r in data['paired_summary']}
    fmt=lambda name,k,m:f"{100*lookup[name,k,m]['test_accuracy_mean']:.2f} ± {100*lookup[name,k,m]['test_accuracy_std']:.2f}"
    result_rows=[]

        for k in (1,2,3,4):

        [LABELS[r['method']],f"{r['improved']}/{r['tied']}/{r['degraded']}",f"{r['mean_delta_pp']:+.2f}",
         f"{r['significant_gain_24']}/{r['significant_decline_24']}"] for r in data['overview']])
    ktable=table(['修改','K1平均Δ pp','K2平均Δ pp','K3平均Δ pp','K4平均Δ pp'],[
        [LABELS[r['method']]]+[f"{r['by_k'][str(k)]:+.2f}" for k in (1,2,3,4)] for r in data['overview']])
    dataset_table=table(['Dataset','LN相对MLP：各K差值pp','GELU相对MLP：各K差值pp'],[
        [name,' / '.join(f"{pairs[name,k,'mlp_ln']['delta_mean_pp']:+.2f}" for k in (1,2,3,4)),
         ' / '.join(f"{pairs[name,k,'mlp_gelu']['delta_mean_pp']:+.2f}" for k in (1,2,3,4))] for name in NAMES])

        [LABELS[r['method']],64 if r['method']=='mlp_ln' else 0,f"{r['mean_fresh_inference_ratio']:.3f}×",
         f"{r['mean_train_step_ratio_historical']:.3f}×",f"{r['mean_peak_memory_delta_mib']:+.3f}"] for r in data['overview']])
    mechanism=[]
    for name in NAMES:

            rows=[lookup[name,k,method] for k in (1,2,3,4)]

                f"{100*np.mean([r['max_weight_over_095_fraction_mean'] for r in rows]):.1f}",
                f"{100*np.mean([r['train_accuracy_mean']-r['test_accuracy_mean'] for r in rows]):.1f}"])
    negative=[]
    for method in ('mlp_ln','mlp_gelu'):
        rows=sorted([r for r in data['paired_summary'] if r['method']==method and r['delta_mean_pp'] < -1e-9],key=lambda r:r['delta_mean_pp'])
        for r in rows[:3]:
            negative.append([LABELS[method],r['dataset'],r['k'],f"{r['delta_mean_pp']:+.2f}",
                f"[{r['ci95_low_pp']:+.2f}, {r['ci95_high_pp']:+.2f}]",f"{r['wins']}/{r['ties']}/{r['losses']}"])

    dot_context='；'.join(f"{LABELS[r['method']]} {r['mean_delta_pp']+mlp_dot_gap:+.2f} pp" for r in data['overview'])
    conclusion=('本协议中观察到部分经各自24项校正后的改善，具体范围见下表；不能将它外推为所有图、所有K的优势。'
        if significant else '本协议未获得经预设各24项多重比较校正后可靠的改善证据；均值差仅作探索性观察，不据此宣称某一修改普遍更好。')
    norm_deltas=[r['layernorm_affine_change_l2'] for r in data['diagnostics'] if r['routing_method']=='mlp_ln']
    assert len(norm_deltas)==240 and min(norm_deltas)>0

## 结构、梯度与基线复现验证

4项组件单元测试通过，包括原MLP的16个CPU/CUDA前向及参数梯度精确对齐子案例、新组件16个不同设备/K的有效梯度检查、R=2首轮score到次轮alpha的反向路径、每次forward logits清零、测试标签替换不改变训练。全部480次新训练中，scorer两个Linear权重均有非零有限梯度，LayerNorm组的gamma权重也有有效梯度；最小首轮裁剪前权重张量梯度范数为{data['minimum_weight_gradient']:.8g}。

LayerNorm的240个模型在最佳Checkpoint中均有仿射参数变化，gamma/beta联合变化L2最小为{min(norm_deltas):.8g}。这证明新增模块参与了训练，不能由此推导它必然改善性能。共享输出标量bias仍可能因softmax平移不变性而梯度为0，这是原结构性质。

在正式Pilot前，Texas K2 split0及Chameleon K2 split0的原MLP各从随机初始化重放一次，训练后权重、预测、log_probs、实际alpha、最佳/停止轮次与旧结果完全相同。随后24个Pilot全部重载核验，再启动Full并复用这24个运行；480矩阵不重复计入锚点或Pilot。主训练失败尝试数为{data['failed_attempts']}，失败文件如有均保留，未隐藏下降结果。

协议哈希 `{verified['protocol_sha256']}`；源模型、factory、scheduler冻结，Full期间没有修改训练核心。旧的{preserved['previous_files_checked']:,}个文件内容/修改时间及原项目{preserved['original_gamma_files_checked']:,}个文件均通过保持不变的复核。
