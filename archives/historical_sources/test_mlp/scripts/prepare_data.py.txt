from pathlib import Path
import sys
sys.path.insert(0,str(Path(__file__).resolve().parents[1]))
from src.paths import ROOT,save_json,sha256
from src.data import DATASETS,SHAPES,load_dataset,load_masks,split_path
import importlib.util
import os
import numpy as np
import torch

def main():
    assert not (ROOT/'data/manifest.json').exists(),'Data preparation already complete; reuse fixed artifacts.'
    spec=importlib.util.spec_from_file_location('official_create_splits',ROOT/'upstream_snapshot/create_splits.py')
    official=importlib.util.module_from_spec(spec);spec.loader.exec_module(official)
    os.chdir(ROOT/'data')
    official.create_splits_directory()
    audit={}
    for name in DATASETS:
        dataset=load_dataset(name,force_reload=True)
        graph=dataset[0]
        assert tuple(graph.x.shape)==SHAPES[name] and graph.y.shape==(graph.num_nodes,)
        assert int(graph.edge_index.min())>=0 and int(graph.edge_index.max())<graph.num_nodes
        assert torch.isfinite(graph.x).all()
        official.create_dataset_splits(name,dataset,num_splits=10)
        records=[]
        for split in range(10):
            masks=load_masks(name,split,graph.num_nodes)
            records.append(dict(split=split,seed=102+split,sha256=sha256(split_path(name,split)),
                                sizes=[int(mask.sum()) for mask in masks],
                                class_counts=[np.bincount(graph.y.numpy()[mask],minlength=dataset.num_classes).tolist() for mask in masks]))
        audit[name]=dict(nodes=graph.num_nodes,features=graph.num_node_features,classes=dataset.num_classes,
                         edges=graph.num_edges,processed_sha256=sha256(dataset.processed_paths[0]),
                         raw_files={str(Path(p).relative_to(ROOT)):sha256(p) for p in dataset.raw_paths},splits=records)
        print('VALIDATED',name,graph.num_nodes,graph.num_edges,flush=True)
    save_json(ROOT/'data/manifest.json',audit)

if __name__=='__main__': main()
