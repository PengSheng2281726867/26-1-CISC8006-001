"""Reload every ablation checkpoint and verify its recorded test predictions."""

import argparse
import csv
import json
from pathlib import Path
import statistics

import run_gate_ablation as experiment
import numpy as np
import torch


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("output", type=Path)
    args = parser.parse_args()
    output = args.output.resolve()
    state = json.loads((output / "results.json").read_text(encoding="utf-8"))
    data_audit = json.loads((output / "data_audit.json").read_text(encoding="utf-8"))
    split_audit = json.loads((output / "split_audit.json").read_text(encoding="utf-8"))
    assert state["status"] == "complete"
    torch.use_deterministic_algorithms(True)
    torch.backends.cuda.matmul.allow_tf32 = False
    torch.backends.cudnn.allow_tf32 = False
    torch.backends.cudnn.deterministic = True
    torch.backends.cudnn.benchmark = False
    for path, digest in state["sources"].items():
        assert experiment.sha256(path) == digest, path
    checkpoints_checked = 0
    result = dict(status="passed", datasets={}, sources_unchanged=True,
                  validation_selected_checkpoints_reloaded=True,
                  exact_test_predictions_reproduced=True)
    for name, pairs in state["pairs"].items():
        assert len(pairs) == 10
        dataset = experiment.load_dataset(name)
        graph = dataset[0].to("cuda")
        assert experiment.sha256(dataset.processed_paths[0]) == data_audit[name]["processed_sha256"]
        for path, digest in data_audit[name]["raw_files"].items():
            assert experiment.sha256(experiment.PROJECT / path) == digest
        for pair, masks_record in zip(pairs, split_audit[name]):
            index = pair["split"]
            assert index == masks_record["seed"] - 102
            mask_path = output / "splits" / masks_record["file"]
            assert experiment.sha256(mask_path) == masks_record["sha256"]
            assert experiment.sha256(experiment.PROJECT / "splits" / mask_path.name) == masks_record["sha256"]
            with np.load(mask_path) as saved:
                masks = {key: saved[key].copy() for key in ("train_mask", "val_mask", "test_mask")}
            assert np.all(np.stack(list(masks.values())).sum(axis=0) == 1)
            for key, mask in masks.items():
                graph[key] = torch.from_numpy(mask).to("cuda")
            assert pair["routing"]["initial_state_sha256"] == pair["uniform"]["initial_state_sha256"]
            assert pair["routing"]["initial_rng_sha256"] == pair["uniform"]["initial_rng_sha256"]
            assert pair["routing"]["parameter_count"] == pair["uniform"]["parameter_count"]
            for variant in ("routing", "uniform"):
                record = pair[variant]
                run_dir = output / record["directory"]
                for filename, key in (("model.pt", "checkpoint_sha256"), ("trace.csv", "trace_sha256"),
                                      ("predictions.npz", "predictions_sha256")):
                    assert experiment.sha256(run_dir / filename) == record[key]
                checkpoint = torch.load(run_dir / "model.pt", map_location="cpu", weights_only=True)
                assert checkpoint["routing"] == (variant == "routing")
                experiment.set_seed(record["seed"])
                model = experiment.GAMMA_GNN(**checkpoint["model_kwargs"]).to("cuda")
                for conv in model.convs:
                    conv.use_routing = checkpoint["routing"]
                assert experiment.state_hash(model.state_dict()) == record["initial_state_sha256"]
                assert experiment.rng_hash() == record["initial_rng_sha256"]
                assert experiment.state_hash(checkpoint["state_dict"]) == record["trained_state_sha256"]
                model.load_state_dict(checkpoint["state_dict"])
                model.eval()
                with torch.no_grad():
                    predictions = model(graph.x, graph.edge_index).argmax(dim=1).cpu().numpy()
                with np.load(run_dir / "predictions.npz") as saved:
                    assert np.array_equal(predictions, saved["predictions"])
                    assert np.array_equal(graph.y.cpu().numpy(), saved["labels"])
                    for key, mask in masks.items():
                        assert np.array_equal(mask, saved[key])
                labels = graph.y.cpu().numpy()
                test_correct = int((predictions[masks["test_mask"]] == labels[masks["test_mask"]]).sum())
                assert test_correct == record["test_correct"]
                assert int(masks["test_mask"].sum()) == record["test_size"]
                assert test_correct / record["test_size"] == record["test_accuracy"]
                val_acc = float((predictions[masks["val_mask"]] == labels[masks["val_mask"]]).mean())
                assert val_acc == record["best_val_accuracy"]
                with (run_dir / "trace.csv").open(newline="", encoding="utf-8") as handle:
                    trace = list(csv.DictReader(handle))
                assert len(trace) == record["stopped_epoch"]
                assert [int(row["epoch"]) for row in trace] == list(range(1, len(trace) + 1))
                assert all(np.isfinite(float(row["train_loss"])) for row in trace)
                values = [float(row["val_accuracy"]) for row in trace]
                assert max(values) == val_acc
                assert values.index(max(values)) + 1 == record["best_epoch"]
                assert record["stopped_epoch"] == 500 or record["stopped_epoch"] - record["best_epoch"] == 50
                checkpoints_checked += 1
        routing = [pair["routing"]["test_accuracy"] for pair in pairs]
        uniform = [pair["uniform"]["test_accuracy"] for pair in pairs]
        summary = state["summary"][name]
        assert abs(statistics.mean(routing) - summary["routing_mean"]) < 1e-12
        assert abs(statistics.mean(uniform) - summary["uniform_mean"]) < 1e-12
        assert abs(statistics.pstdev(routing) - summary["routing_std_ddof0"]) < 1e-12
        assert abs(statistics.pstdev(uniform) - summary["uniform_std_ddof0"]) < 1e-12
        difference = [100 * (a - b) for a, b in zip(routing, uniform)]
        assert abs(statistics.mean(difference) - summary["mean_difference_pp"]) < 1e-12
        assert 0 <= summary["exact_sign_flip_p_two_sided"] <= summary["holm_adjusted_p"] <= 1
        result["datasets"][name] = dict(pairs_checked=10, checkpoints_checked=20,
                                        mean_difference_pp=summary["mean_difference_pp"])
        print(f"VERIFIED {name}: 20 checkpoints, initial states, exact predictions, best-validation selection and statistics", flush=True)
    result["checkpoints_checked"] = checkpoints_checked
    experiment.write_json(output / "verification.json", result)
    print(f"COMPLETE: {checkpoints_checked} checkpoint evaluations reproduced exactly", flush=True)


if __name__ == "__main__":
    main()
