EVIDENCE=[]

def toy_graph():
    seed_all(30)
    return Data(x=torch.randn(8,5),edge_index=torch.tensor([[0,1,2,3,4,5,6,7,1,4],[1,2,3,4,5,6,7,0,4,1]]),
                y=torch.tensor([0,1,2,0,1,2,0,1]),train_mask=torch.tensor([1,1,1,1,0,0,0,0],dtype=torch.bool),
                val_mask=torch.tensor([0,0,0,0,1,1,0,0],dtype=torch.bool),
                test_mask=torch.tensor([0,0,0,0,0,0,1,1],dtype=torch.bool))

class MLPTests(unittest.TestCase):
    def test_shapes_gradient_and_k(self):
        for device in ('cpu','cuda') if torch.cuda.is_available() else ('cpu',):
            for k in (1,2,3,4):
                with self.subTest(device=device,k=k):
                    graph=toy_graph().to(device);seed_all(12)
                    model=GammaNet(5,32,3,powers=list(range(k+1)),routing_score='mlp').to(device)
                    captured=[]
                    hook=model.convs[0].scorer[0].register_forward_pre_hook(lambda module,inputs:captured.append(tuple(inputs[0].shape)))
                    out,details=model(graph.x,graph.edge_index,return_diagnostics=True)
                    hook.remove()
                    self.assertEqual(tuple(out.shape),(8,3))
                    self.assertEqual(captured,[(8,k+1,64)]*2)
                    first=details[0]['iterations'][0]['agreement'];first.retain_grad()
                    torch.nn.functional.nll_loss(out[graph.train_mask],graph.y[graph.train_mask]).backward()
                    self.assertGreater(float(first.grad.abs().max()),0)
                    gradients={name:float(p.grad.norm()) if p.grad is not None else None for name,p in model.named_parameters() if '.scorer.' in name}
                    for name,p in model.named_parameters():
                        if p.grad is not None: self.assertTrue(bool(torch.isfinite(p.grad).all()),name)
                        if '.scorer.' in name and name.endswith('weight'): self.assertGreater(float(p.grad.norm()),0,name)
                    EVIDENCE.append(dict(device=device,k=k,mlp_input_shape=captured[0],gradient_norms=gradients))

    def test_propagation_range(self):
        edge=torch.stack((torch.arange(5),torch.arange(1,6)))
        x=torch.zeros(6,2);x[0,0]=1
        for k in (1,2,3,4):
            layer=GammaLayer(2,2,powers=list(range(k+1)),routing_score='dot')
            with torch.no_grad():
                layer.shared_linear.weight.copy_(torch.eye(2));layer.shared_linear.bias.zero_();layer.scale_params.fill_(1)
            with patch.object(layer,'propagate',wraps=layer.propagate) as spy:
                hops=layer.hop_embeddings(x,edge)
            self.assertEqual(spy.call_count,k*(k+1)//2)
            self.assertGreater(float(hops[k,-1].norm()),0)
            self.assertEqual(float(hops[k+1].abs().max()),0)

    def test_second_iteration_and_reset(self):
        graph=toy_graph();layer=GammaLayer(5,32,routing_score='mlp').eval()
        out,details=layer(graph.x,graph.edge_index,return_diagnostics=True)
        alpha=details['iterations'][1]['alpha']
        self.assertFalse(torch.allclose(alpha,torch.full_like(alpha,1/3)))
        expected=layer.squash((alpha.unsqueeze(-1)*details['hops']).sum(1))+layer.bias
        torch.testing.assert_close(out,expected)
        layer.num_iterations=1;one=layer(graph.x,graph.edge_index)
        self.assertFalse(torch.allclose(one,out))
        one.sum().backward()
        self.assertTrue(all(p.grad is None for p in layer.scorer.parameters()))
        layer.num_iterations=2;layer.zero_grad();again,details2=layer(graph.x,graph.edge_index,return_diagnostics=True)
        torch.testing.assert_close(out,again,rtol=0,atol=0)
        self.assertEqual(float(details2['iterations'][0]['logits'].abs().max()),0)
        self.assertFalse(any('logit' in key for key in layer.state_dict()))

    def test_matched_initialization(self):
        expected_common=None;expected_scorer=None;expected_rng=None
        for k in (1,2,3,4):
            for method in ('dot','mlp'):
                model,meta=build_model(5,3,k,method,42,'cpu')
                expected_common=expected_common or meta['common_init_sha256']
                expected_rng=expected_rng or meta['cpu_rng_sha256']
                self.assertEqual(meta['common_init_sha256'],expected_common)
                self.assertEqual(meta['cpu_rng_sha256'],expected_rng)
                if method=='mlp':
                    expected_scorer=expected_scorer or meta['scorer_init_sha256']
                    self.assertEqual(meta['scorer_init_sha256'],expected_scorer)
                    self.assertEqual(sum(p.numel() for n,p in model.named_parameters() if '.scorer.' in n),2113)

    def test_shared_real_splits_and_no_overlap(self):
        for name in DATASETS:
            dataset=load_dataset(name)
            for split in range(10):
                dot=load_masks(name,split,dataset[0].num_nodes);mlp=load_masks(name,split,dataset[0].num_nodes)
                for first,second in zip(dot,mlp): np.testing.assert_array_equal(first,second)

    def test_test_labels_do_not_affect_training(self):
        config=json.loads((ROOT/'configs/main.json').read_text());config['epochs']=8
        for method in ('dot','mlp'):
            graph=toy_graph();corrupted=graph.clone();corrupted.y[corrupted.test_mask]=(corrupted.y[corrupted.test_mask]+1)%3
            first,_=build_model(5,3,2,method,42,'cpu');second=copy.deepcopy(first)
            seed_all(97);report1=train_model(first,graph,config)
            seed_all(97);report2=train_model(second,corrupted,config)
            self.assertEqual(tensor_hash(first.state_dict()),tensor_hash(second.state_dict()))
            self.assertEqual(report1['best_validation_epoch'],report2['best_validation_epoch'])
            self.assertEqual([r['train_loss'] for r in report1['history']],[r['train_loss'] for r in report2['history']])
