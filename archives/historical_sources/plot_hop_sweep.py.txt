"""Export a standard static accuracy-versus-K figure."""

import argparse
import json
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("output", type=Path)
    args = parser.parse_args()
    root = args.output.resolve()
    state = json.loads((root / "results.json").read_text(encoding="utf-8"))
    assert state["status"] == "complete"
    names = ["Texas", "Wisconsin", "Cornell", "Actor", "Squirrel", "Chameleon"]
    plt.rcParams.update({"font.size": 10, "axes.spines.top": False, "axes.spines.right": False,
                         "svg.fonttype": "none", "font.family": "DejaVu Sans"})
    fig, axes = plt.subplots(2, 3, figsize=(11.5, 6.5))
    for axis, name in zip(axes.flat, names):
        rows = sorted([row for row in state["summary"] if row["dataset"] == name], key=lambda row: row["k"])
        x = [row["k"] for row in rows]
        for prefix, label, color, marker in (("routing", "Dynamic routing", "#2264A5", "o"),
                                               ("uniform", "Uniform aggregation", "#B65E28", "s")):
            means = [100 * row[f"{prefix}_mean"] for row in rows]
            stds = [100 * row[f"{prefix}_std_ddof0"] for row in rows]
            axis.plot(x, means, marker=marker, color=color, linewidth=1.8, markersize=5, label=label)
            axis.fill_between(x, [a-b for a,b in zip(means,stds)], [a+b for a,b in zip(means,stds)],
                              color=color, alpha=0.10, linewidth=0)
        axis.set_title(name, fontweight="bold")
        axis.set_xticks([2, 3, 4])
        axis.set_xlim(1.85, 4.15)
        axis.grid(axis="y", alpha=0.18)
    for axis in axes[-1]:
        axis.set_xlabel("Maximum propagation power K")
    for axis in axes[:, 0]:
        axis.set_ylabel("Test accuracy (%)")
    handles, labels = axes[0, 0].get_legend_handles_labels()
    fig.legend(handles, labels, loc="upper center", ncol=2, frameon=False, bbox_to_anchor=(0.5, 0.94))
    fig.suptitle("Maximum-hop sweep: dynamic routing versus uniform aggregation", fontsize=14, y=0.99)
    fig.text(0.5, 0.015, f"Candidates: {state['hop_convention']}  |  10 paired splits per configuration  |  Shading: ±1 standard deviation",
             ha="center", fontsize=9, color="#444444")
    fig.tight_layout(rect=(0, 0.045, 1, 0.9), h_pad=2, w_pad=2)
    fig.savefig(root / "accuracy_vs_k.png", dpi=180)
    fig.savefig(root / "accuracy_vs_k.svg")
    print(f"Saved {root / 'accuracy_vs_k.png'}")


if __name__ == "__main__":
    main()
